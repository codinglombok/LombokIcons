Get Started

Pick the integration method that fits your stack.

<!-- CSS -->
<link rel="stylesheet"
      href="https://cdn.jsdelivr.net/npm/lombokicons/dist/css/lombokicons.min.css">

<!-- Icons via data attributes -->
<i data-lf="home"></i>
<i data-lf="search" data-lf-mode="duo"></i>
<i data-lf="bell" data-lf-mode="bold" class="lf-lg lf-ring"></i>

<!-- Auto-replace script -->
<script type="module">
  import { replaceIcons } from
    'https://cdn.jsdelivr.net/npm/lombokicons/dist/js/lombokicons.mjs';
  replaceIcons();
</script>
